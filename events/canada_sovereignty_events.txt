namespace = canada_sovereignty

# The opening question, fired in the first month
canada_sovereignty.1 = {
	type = country_event
	placement = root

	title = canada_sovereignty.1.t
	desc = canada_sovereignty.1.d
	flavor = canada_sovereignty.1.f

	event_image = {
		video = "unspecific_signed_contract"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	duration = 3

	immediate = {
		set_variable = canada_sovereignty_fired
		# The journal entry is added no matter what you pick. It closes
		# itself once Canada is independent, so it only lingers if you
		# wait or if a war of independence fails.
		add_journal_entry = { type = je_canadian_sovereignty }
	}

	option = {
		name = canada_sovereignty.1.a
		ai_chance = { base = 40 }
	}

	option = {
		name = canada_sovereignty.1.b
		default_option = yes
		canada_request_accord = yes
		ai_chance = { base = 50 }
	}

	option = {
		name = canada_sovereignty.1.c
		canada_hard_break = yes
		ai_chance = { base = 10 }
	}
}

# Terms of the accord, reached from the event or the journal button
canada_sovereignty.2 = {
	type = country_event
	placement = root

	title = canada_sovereignty.2.t
	desc = canada_sovereignty.2.d
	flavor = canada_sovereignty.2.f

	trigger = {
		NOT = { has_variable = canada_hard_break_started }
	}

	event_image = {
		video = "unspecific_signed_contract"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	duration = 3

	# Standard accord. Two copies so the recommended tick lands on it only
	# while the Newfoundland option is locked.
	option = {
		name = canada_sovereignty.2.a
		trigger = {
			c:GBR = { owns_entire_state_region = STATE_NEWFOUNDLAND }
			je:je_canadian_sovereignty ?= {
				"scripted_bar_progress(canada_national_identity_bar)" >= 50
			}
		}
		canada_negotiated_independence = yes
		ai_chance = { base = 40 }
	}
	option = {
		name = canada_sovereignty.2.a
		default_option = yes
		trigger = {
			NAND = {
				c:GBR = { owns_entire_state_region = STATE_NEWFOUNDLAND }
				je:je_canadian_sovereignty ?= {
					"scripted_bar_progress(canada_national_identity_bar)" >= 50
				}
			}
		}
		canada_negotiated_independence = yes
		ai_chance = { base = 40 }
	}

	# Accord plus Newfoundland, at the cost of some goodwill. London only
	# gives ground to a Canada with a strong sense of itself.
	option = {
		name = canada_sovereignty.2.b
		default_option = yes
		trigger = {
			c:GBR = { owns_entire_state_region = STATE_NEWFOUNDLAND }
			je:je_canadian_sovereignty ?= {
				"scripted_bar_progress(canada_national_identity_bar)" >= 50
			}
		}
		canada_gain_newfoundland = yes
		canada_negotiated_independence = yes
		change_relations = {
			country = c:GBR
			value = -20
		}
		ai_chance = { base = 60 }
	}

	# Locked copy of the Newfoundland option, shown greyed out below 50
	option = {
		name = canada_sovereignty.2.b
		trigger = { always = no }
		show_as_unavailable = {
			c:GBR = { owns_entire_state_region = STATE_NEWFOUNDLAND }
			NOT = {
				je:je_canadian_sovereignty ?= {
					"scripted_bar_progress(canada_national_identity_bar)" >= 50
				}
			}
		}
		custom_tooltip = canada_identity_newfoundland_tt
		ai_chance = { base = 0 }
	}

	# Walk away for now; the journal entry stays open
	option = {
		name = canada_sovereignty.2.c
		ai_chance = { base = 0 }
	}
}

# The Confederation Conference, fired to Upper Canada on day one.
# The host city becomes the capital; declining keeps the base game start.
canada_sovereignty.3 = {
	type = country_event
	placement = root

	title = canada_sovereignty.3.t
	desc = canada_sovereignty.3.d
	flavor = canada_sovereignty.3.f

	event_image = {
		video = "unspecific_signed_contract"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	duration = 3

	# Ottawa: the capital stays in Ontario, and Bytown takes its new name
	option = {
		name = canada_sovereignty.3.a
		default_option = yes
		canada_early_confederation_setup = yes
		s:STATE_ONTARIO = {
			random_scope_state = {
				limit = { owner = root }
				set_hub_name = {
					type = city
					name = HUB_NAME_STATE_ONTARIO_city_ottawa
				}
			}
		}
		ai_chance = { base = 70 }
	}

	# Quebec City
	option = {
		name = canada_sovereignty.3.b
		canada_early_confederation_setup = yes
		set_capital = STATE_QUEBEC
		ai_chance = { base = 20 }
	}

	# Halifax
	option = {
		name = canada_sovereignty.3.c
		canada_early_confederation_setup = yes
		set_capital = STATE_NEW_BRUNSWICK
		ai_chance = { base = 10 }
	}

	# Decline: British North America stays divided, as in the base game
	option = {
		name = canada_sovereignty.3.e
		ai_chance = { base = 0 }
	}
}

# London's answer to Canada's request for negotiations
canada_sovereignty.8 = {
	type = country_event
	placement = root

	title = canada_sovereignty.8.t
	desc = canada_sovereignty.8.d
	flavor = canada_sovereignty.8.f

	event_image = {
		video = "unspecific_politicians_arguing"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	duration = 3

	# Moot if Canada has taken up arms since asking
	trigger = {
		exists = c:CAN
		c:CAN = { NOT = { has_variable = canada_hard_break_started } }
	}

	# Receive the delegation; Canada then chooses its terms
	option = {
		name = canada_sovereignty.8.a
		default_option = yes
		c:CAN = {
			remove_variable = canada_accord_pending
			trigger_event = { id = canada_sovereignty.2 }
		}
		ai_chance = {
			base = 75
			modifier = {
				trigger = { c:CAN ?= { relations:root < relations_threshold:cordial } }
				add = -50
			}
		}
	}

	# Send them home; Canadians will not forget it
	option = {
		name = canada_sovereignty.8.b
		c:CAN = {
			remove_variable = canada_accord_pending
			set_variable = { name = canada_accord_refused days = 1825 }
			trigger_event = { id = canada_sovereignty.9 }
		}
		ai_chance = { base = 25 }
	}
}

# Canada weighs Washington's offer, and its price
canada_sovereignty.11 = {
	type = country_event
	placement = root

	title = canada_sovereignty.11.t
	desc = canada_sovereignty.11.d
	flavor = canada_sovereignty.11.f

	event_image = {
		video = "unspecific_signed_contract"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	duration = 3

	trigger = {
		exists = c:USA
	}

	# American arms and money, paid for with American access to our markets
	option = {
		name = canada_sovereignty.11.a
		default_option = yes
		add_modifier = { name = canada_american_backing days = 1825 }
		create_treaty = {
			first_country = root
			second_country = c:USA
			is_draft = no
			binding_period = { years = 10 }
			articles_to_create = {
				{
					article = trade_privilege
					source_country = root
					target_country = c:USA
				}
			}
		}
		change_relations = { country = c:USA value = 20 }
		change_relations = { country = c:GBR value = -10 }
		ai_chance = { base = 70 }
	}

	# Canada stands on its own
	option = {
		name = canada_sovereignty.11.b
		je:je_canadian_sovereignty ?= {
			add_progress = { value = 5 name = canada_national_identity_bar }
		}
		ai_chance = { base = 30 }
	}
}
