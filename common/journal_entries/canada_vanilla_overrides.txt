# Copy of vanilla je_canada_can (common/journal_entries/00_canada_australia.txt,
# game 1.13.11). This file loads after vanilla, so this definition wins.
# Changes: hidden and invalid once the Confederation Conference has united
# Canada, since the Path to Sovereignty takes over from there.
# Re-copy it from vanilla if a patch changes the original.

je_canada_can = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_historical_content

	is_shown_when_inactive = {
		NOT = { has_variable = canada_confederated }
		capital = {
			is_in_geographic_region = geographic_region_greater_canada
		}
		is_subject = yes
	}

	is_shown_in_lobby = {
		OR = {
			country_has_primary_culture = cu:anglo_canadian
			country_has_primary_culture = cu:franco_canadian
		}
		is_in_geographic_region = geographic_region_greater_canada
	}

	possible = {
		NOT = { has_variable = canada_confederated }
		has_technology_researched = nationalism
	}

	scripted_button = je_canada_unite_can_button

	complete = {
		any_state_in_greater_canada = {
			filter = {
				NOR = {
					this.state_region = s:STATE_WASHINGTON
					this.state_region = s:STATE_IDAHO
					this.state_region = s:STATE_OREGON
					this.state_region = s:STATE_NEWFOUNDLAND
				}
			}
			owner = ROOT
			percent = 1
		}
	}

	on_complete = {
		trigger_event = { id = can_aus.5 }
		hidden_effect = {
			if = {
				limit = {
					any_scope_state = {
						state_region = s:STATE_ONTARIO
					}
				}
			}
			s:STATE_ONTARIO = {
				random_scope_state = {
			        set_hub_name = {
			            type = city
			            name = HUB_NAME_STATE_ONTARIO_city_ottawa
			        }
				}
			}
		}
	}

	invalid = {
		OR = {
			is_subject = no
			has_variable = canada_confederated
		}
	}

	weight = 100


	should_be_pinned_by_default_uninvolved_or_context = yes
}
