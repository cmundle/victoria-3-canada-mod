# Replaces the vanilla file of the same name. Vanilla keeps the first
# definition of a key, so je_canada_can can only be changed by replacing the
# whole file. The only intended change is to je_canada_can: it hides and goes
# invalid once the Confederation Conference unites Canada (variable
# canada_confederated). Everything else should match vanilla.
#
# Patch 1.15 merged complete, on_complete and invalid into outcome blocks, and
# this file was converted by hand from the 1.13.11 original. Re-copy it from
# the current vanilla file if a patch changes the original again (see the
# modding notes in the README).

je_canada_can = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_historical_content

	is_shown_when_inactive = {
		NOT = { has_variable = canada_confederated }
		capital = {
			is_in_geographic_region = geographic_region_greater_canada
		}
		is_subject = yes
	}

	is_shown_in_lobby = {
		OR = {
			country_has_primary_culture = cu:anglo_canadian
			country_has_primary_culture = cu:franco_canadian
		}
		is_in_geographic_region = geographic_region_greater_canada
	}

	possible = {
		NOT = { has_variable = canada_confederated }
		has_technology_researched = nationalism
	}

	scripted_button = je_canada_unite_can_button

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = canada_mod_je_canada_can_completed_desc
		trigger = {
			any_state_in_greater_canada = {
				filter = {
					NOR = {
						this.state_region = s:STATE_WASHINGTON
						this.state_region = s:STATE_IDAHO
						this.state_region = s:STATE_OREGON
						this.state_region = s:STATE_NEWFOUNDLAND
					}
				}
				owner = ROOT
				percent = 1
			}
		}
		effect = {
			trigger_event = { id = can_aus.5 }
			hidden_effect = {
				if = {
					limit = {
						any_scope_state = {
							state_region = s:STATE_ONTARIO
						}
					}
				}
				s:STATE_ONTARIO = {
					random_scope_state = {
				        set_hub_name = {
				            type = city
				            name = HUB_NAME_STATE_ONTARIO_city_ottawa
				        }
					}
				}
			}
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = canada_mod_je_canada_can_invalid_desc
		trigger = {
			OR = {
				is_subject = no
				has_variable = canada_confederated
			}
		}
		effect = {}
	}

	weight = 100


	should_be_pinned_by_default_uninvolved_or_context = yes
}
