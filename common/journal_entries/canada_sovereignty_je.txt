je_canadian_sovereignty = {
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	group = je_group_foreign_affairs

	scripted_progress_bar = canada_national_identity_bar

	scripted_button = canada_negotiate_accord_button
	scripted_button = canada_approach_washington_button
	scripted_button = canada_break_hard_button

	# Completes however independence happens (accord or winning the war)
	complete = {
		is_subject = no
	}

	# An independent Canada turns west: Frontier Colonization, unless a
	# colonization law is already in place
	on_complete = {
		if = {
			limit = { has_variable = canada_hard_break_started }
			trigger_event = { id = canada_sovereignty.6 }
		}
		# Washington's price, due now that Canada has its own market
		if = {
			limit = {
				has_variable = canada_owes_washington
				exists = c:USA
			}
			remove_variable = canada_owes_washington
			create_treaty = {
				first_country = root
				second_country = c:USA
				is_draft = no
				binding_period = { years = 10 }
				articles_to_create = {
					{
						article = trade_privilege
						source_country = root
						target_country = c:USA
					}
				}
			}
		}
		# The Oregon bargain: the promised alliance, once Canada can sign one
		if = {
			limit = {
				has_variable = canada_us_alliance_promised
				exists = c:USA
			}
			remove_variable = canada_us_alliance_promised
			create_treaty = {
				first_country = root
				second_country = c:USA
				is_draft = no
				binding_period = { years = 99 }
				articles_to_create = {
					{ article = alliance }
				}
			}
		}
		add_technology_researched = colonization
		if = {
			limit = { has_law = law_type:law_no_colonial_affairs }
			activate_law = law_type:law_frontier_colonization
		}
	}

	# While Canada waits, London and the reformers both push
	on_monthly_pulse = {
		random_events = {
			190 = 0
			5 = canada_sovereignty.4
			5 = canada_sovereignty.5
		}
	}

	invalid = {
		NOT = { exists = c:GBR }
	}

	weight = 100

	should_be_pinned_by_default_involved = yes
	should_be_pinned_by_default_uninvolved_or_context = yes
}
