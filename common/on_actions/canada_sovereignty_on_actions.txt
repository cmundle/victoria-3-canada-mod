# These merge with the vanilla on_action lists. Only on_actions and events
# blocks may be added to vanilla on_actions.

# Runs once the player has picked a country in the lobby
on_game_started_after_lobby = {
	on_actions = { canada_early_confederation }
}

on_monthly_pulse_country = {
	on_actions = { canada_setup_fallback canada_sovereignty_check canada_hard_break_failed }
}

on_yearly_pulse_country = {
	on_actions = { canada_accord_upkeep }
}

canada_early_confederation = {
	trigger = {
		NOT = { has_global_variable = canada_setup_done }
		NOT = { has_game_rule = canada_start_vanilla }
	}
	effect = {
		debug_log = "CANADA MOD: conference called from lobby hook"
		set_global_variable = canada_setup_done
		c:ONT ?= { trigger_event = { id = canada_sovereignty.3 } }
	}
}

# Fallback: if the lobby hook never fired, Upper Canada gets the
# conference on its first monthly pulse instead
canada_setup_fallback = {
	trigger = {
		exists = c:ONT
		this = c:ONT
		NOT = { has_global_variable = canada_setup_done }
		NOT = { has_game_rule = canada_start_vanilla }
	}
	effect = {
		debug_log = "CANADA MOD: conference called from monthly fallback"
		set_global_variable = canada_setup_done
		trigger_event = { id = canada_sovereignty.3 }
	}
}

# Backup: fires the sovereignty event if the setup's own trigger was missed
canada_sovereignty_check = {
	trigger = {
		exists = c:CAN
		this = c:CAN
		is_subject_of = c:GBR
		NOT = { has_variable = canada_sovereignty_fired }
	}
	effect = {
		trigger_event = { id = canada_sovereignty.1 }
	}
}

# Standing friendship after the accord: Britain's relations with Canada
# are topped up every year so the AI never drifts back to hostility
canada_accord_upkeep = {
	trigger = {
		exists = c:CAN
		this = c:GBR
		c:CAN = { has_variable = canada_accord_signed }
	}
	effect = {
		change_relations = { country = c:CAN value = 15 }
	}
}

# A hard break that ends with Canada still a subject: the war was lost
# or the play backed down. Independence by force is caught by the
# journal entry's on_complete instead.
canada_hard_break_failed = {
	trigger = {
		has_variable = canada_hard_break_started
		is_subject_of = c:GBR
		is_at_war = no
		is_diplomatic_play_initiator = no
	}
	effect = {
		remove_variable = canada_hard_break_started
		trigger_event = { id = canada_sovereignty.7 }
	}
}
