# National identity on the Path to Sovereignty. Fills month by month as
# Canada grows literate, prosperous and self-governing; events push it
# up or down. The accord's terms get better as it rises.
canada_national_identity_bar = {
	name = "canada_national_identity_bar"
	desc = "canada_national_identity_bar_desc"

	# Patch 1.15 removed the ready-made default_green look, so the green
	# bar is the neutral white art tinted green.
	progress_texture = "gfx/interface/progressbar/progressbar_white.dds"
	glow_texture = "gfx/interface/progressbar/progressbar_white_glow.dds"
	glow_line_texture = "gfx/interface/progressbar/progressbar_white_glow_line.dds"
	color = { 0.35 0.75 0.4 1 }

	monthly_progress = {
		add = {
			desc = "canada_identity_generation_tt"
			value = 0.1
		}
		if = {
			limit = { literacy_rate >= 0.3 }
			add = {
				desc = "canada_identity_literacy_tt"
				value = 0.15
			}
		}
		if = {
			limit = { country_has_voting_franchise = yes }
			add = {
				desc = "canada_identity_franchise_tt"
				value = 0.15
			}
		}
		if = {
			limit = { gdp_ranking <= 20 }
			add = {
				desc = "canada_identity_gdp_tt"
				value = 0.15
			}
		}
		if = {
			limit = { sol_ranking <= 20 }
			add = {
				desc = "canada_identity_sol_tt"
				value = 0.15
			}
		}
		if = {
			limit = { country_rank >= rank_value:major_power }
			add = {
				desc = "canada_identity_rank_tt"
				value = 0.15
			}
		}
	}

	start_value = 20
	min_value = 0
	max_value = 100
}
